get_filename_component(BRICKPHYR_ROOT ../.. ABSOLUTE)
set(KCONFIG_ROOT ${BRICKPHYR_ROOT}/Kconfig)
set(CONF_FILE ../arm.conf ../native/prj.conf prj.conf storage.conf)
list(APPEND DTS_ROOT ${BRICKPHYR_ROOT} ${BRICKPHYR_ROOT}/tests/ports)
list(APPEND SYSCALL_INCLUDE_DIRS ${BRICKPHYR_ROOT}/sdk/include ${BRICKPHYR_ROOT}/include)
find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})
project(native_smp)
include(${CMAKE_CURRENT_SOURCE_DIR}/../../cmake/pbio.cmake)
if(CONFIG_BRICKPHYR_MOTOR_CONTROL)
  include(${BRICKPHYR_ROOT}/cmake/motor_control.cmake)
endif()
if(NOT CONFIG_SHELL_BACKEND_SERIAL)
  target_sources(app PRIVATE ${BRICKPHYR_ROOT}/src/host_transport.c
    ${BRICKPHYR_ROOT}/src/host_protocol.c
    ${WEST_TOPDIR}/pybricks-micropython/lib/pbio/src/protocol/pybricks.c)
  target_include_directories(app PRIVATE ${WEST_TOPDIR}/pybricks-micropython/lib/pbio/include)
endif()
include(${BRICKPHYR_ROOT}/sdk/native.cmake)
set(native_probes record record_event record_paused record_variable record_frame control control_file)
if(CONFIG_BRICKPHYR_PYTHON)
  list(APPEND native_probes capacity)
endif()
if(CONFIG_BRICKPHYR_IMU)
  list(APPEND native_probes control_imu)
  if(CONFIG_SETTINGS_FILE)
    list(APPEND native_probes settings_save)
  endif()
endif()
foreach(probe IN LISTS native_probes)
  add_llext_target(${probe}_probe OUTPUT ${PROJECT_BINARY_DIR}/${probe}_probe.elf
    SOURCES ${BRICKPHYR_ROOT}/tests/native/${probe}.c)
  add_dependencies(${probe}_probe_llext_lib brickphyr_api)
  llext_compile_options(${probe}_probe -O2 -g0)
  llext_link_options(${probe}_probe -nostdlib -Wl,-T,${BRICKPHYR_ROOT}/sdk/include/brickphyr.ld.h)
  get_target_property(probe_target ${probe}_probe lib_target)
  if(probe STREQUAL "capacity")
    # Admission/MPU-span check; the native lifecycle fixture sweeps every byte.
    target_compile_definitions(${probe_target} PRIVATE CAPACITY_STACK_SIZE=8192 CAPACITY_STRIDE=1024)
  endif()
  target_link_libraries(${probe_target} PRIVATE c gcc)
  add_dependencies(app ${probe}_probe)
endforeach()
generate_inc_file(${PROJECT_BINARY_DIR}/control_probe.elf
  ${PROJECT_BINARY_DIR}/background_controller.inc)
add_custom_target(background_controller DEPENDS ${PROJECT_BINARY_DIR}/background_controller.inc)
foreach(mode RANGE 0 2)
  set(probe background_${mode})
  add_llext_target(${probe} OUTPUT ${PROJECT_BINARY_DIR}/${probe}.elf
    SOURCES ${BRICKPHYR_ROOT}/tests/native/background.c)
  add_dependencies(${probe}_llext_lib brickphyr_api background_controller)
  target_include_directories(${probe}_llext_lib PRIVATE ${PROJECT_BINARY_DIR})
  target_compile_definitions(${probe}_llext_lib PRIVATE BACKGROUND_MODE=${mode})
  llext_compile_options(${probe} -O2 -g0)
  llext_link_options(${probe} -nostdlib -Wl,-T,${BRICKPHYR_ROOT}/sdk/include/brickphyr.ld.h)
  target_link_libraries(${probe}_llext_lib PRIVATE c gcc)
  add_dependencies(app ${probe})
endforeach()
